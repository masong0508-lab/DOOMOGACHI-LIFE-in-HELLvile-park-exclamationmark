# coresave1.txt - the core's streamed saves on the 128 KB flash, first
# run (fresh save file). make -C gba test-coresave runs coresave2.txt next
# with the same --save file. Commands: see harness.c (mailbox, src/hcmd.c).
wait 40
expect-log doomo: init
# the story runner commands (SPEC R20, hcmd.c): a pinned seed, a story flag
seed 4242
expect-flag 0 0
# the codec's worst case (GBA-500: 500 random custom citizens and 1024 edges, ~31 KB; it was
# ~58 KB with 1000), slot 1 (crosses the 64 KB bank)
town-fill 4242
town-crc
save 1
expect-save 1
verify 1
# GBA-500: it fits the 40 KB story slot too now (with 1000 it was refused there)
save 0
verify 0
# a typical story island (premade citizens, lived in) in slot 0
town-fill 0x80000001
town-crc
save 0
town-fill 7
load 0
expect-town-crc
# and the big one back from slot 1
town-fill 4242
town-crc
town-fill 7
load 1
expect-town-crc
wait 10
