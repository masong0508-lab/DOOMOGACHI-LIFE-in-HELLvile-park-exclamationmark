# busy_banner.txt - the pixels around the "Saving settings..." banner (PS1-RAM step 3: gfx_busy_banner()
# keeps the strip under the banner in VRAM, not in a .bss copy). Shots (VRAM) and dshots (the displayed
# picture) 0, 1, 2, 4 and 24 frames after B, twice. Play it on two builds and compare the PNGs byte for byte
# (docs/PS1.md "RAM headroom 2026-11-12", "Checks"):
#   python3 ps1/harness/run.py --exe ps1/build/doomogachi.exe --cue ps1/build/doomogachi.cue \
#       --script ps1/harness/scripts/busy_banner.txt --out OUT --expect "busy_banner: done"

scene TITLE

wait 30

wait 10

shot b01_after_B
wait 1
dshot d02
shot b02
wait 1
dshot d03
shot b03
wait 2
dshot d04
shot b04
wait 20
dshot d05
shot b05
# second visit, in-place rewrite

wait 30

press LEFT
wait 10
press B
dshot d11_after_B
shot b11_after_B
wait 3
dshot d12
