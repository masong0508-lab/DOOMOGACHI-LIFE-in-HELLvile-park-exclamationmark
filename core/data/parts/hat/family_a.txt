# core/data/parts/hat/family_a.txt - the hat family, first batch (CR2): 12 hats built from the macro
# ops (crown, brim, band, bow, ring, chain, frill) and three templates. Hats are drawn by ph_hat's
# hook in group 5; `scale 128` doubles every coordinate so a crown can reach 2.4 head radii up.
# They sit after the 24 painted hats (values 24..35) and are not selectable until the hat list is
# flipped live (state=hidden in core/data/parts/catalog/hat.txt).

# a hat in the hat colour: the crown, a brim, and a band one tone darker (the template arguments
# are the crown and brim kinds and their sizes)
def hat_cb CK CY CW CH BK BW BH
  mat hat_color
  shade ball
  solid
  brim $BK 0 $CY $BW $BH
  solid
  crown $CK 0 $CY $CW $CH
enddef

def hat_band Y W H
  mat hat_color dark
  band 0 $Y $W $H
enddef

# a bow on the hat's side in the accessory colour
def side_bow X Y S
  mat accessory_color
  shade ball
  solid
  bow $X $Y $S
enddef

list hat 24                      # Gumdrop Hat: a sugar dome with a frosting frill
block group=5 anchor=head lod=all view=all
  scale 128
  mat hat_color
  shade gloss
  solid
  crown dome 0 -88 132 104
  mat hat_color light
  solid
  frill -112 112 -92 6 18
  ifview portrait,figure
  mat accessory_color
  decal
  dot -40 -150 2
  dot 30 -170 2
  dot 70 -130 2
  dot -80 -120 2
  endif
end

list hat 25                      # Stovepipe: a very tall cylinder on a wide brim
block group=5 anchor=head lod=all view=all
  scale 128
  use hat_cb cylinder -96 58 86 wide 104 18
  use hat_band -112 58 10
end
