# core/data/home/themes.txt - the house's 10 themes (owner: H1; HF sec.2.3).
# furngen (H4) joins this into core/gen/furn_data.c (doomo_home_themes[]).
#
# "Apply theme" repaints the walls and the floor, tints the ceiling and
# recolours every free paint slot (a slot still on its catalog default) to
# the theme's two paints. It is one undo step.
#
#   [theme THEME_ID] "Name (<= 20)"
#     walls PATTERN PAINT     wallpaper pattern 0..31 and its paint
#     floor PATTERN PAINT     floor pattern 0..31 and its paint
#     paints A B              what free paint slots A / B become
#     ceiling N               ceiling tint 0..7 (the cornice; PS1: the ceiling)
#     accent PAINT            trims, frames, the HUD's highlight in the room
#
# Wallpaper patterns: 0 stripes, 1 dots, 2 wainscot, 3 panels, 4 hearts,
# 5 harlequin, 6 stone, 7 stars. Floors: 0 checker, 1 planks, 2 tiles,
# 3 herringbone, 4 grass, 5 flagstone, 6 lava crack, 7 cloud. (The
# renderers take pattern & 7; the rest of the range is reserved.)
